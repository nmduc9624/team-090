# Generic Password Spray Hunt
telemetry: auth
fields: timestamp, user, src_ip, user_agent, application, action, status, failure_reason
logic:
  from auth within last 1h
  where action is login and status is failure
  group by src_ip, application, 15m window
  calculate distinct failed users, total failures, success count after failures
  alert when distinct failed users is high and failures per user are low
correlate:
  check successful login from the same src_ip/user_agent after the spray window.
