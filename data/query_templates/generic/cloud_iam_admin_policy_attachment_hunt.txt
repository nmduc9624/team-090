# Generic Cloud IAM Admin Policy Attachment Hunt
telemetry: cloud_audit, auth
fields: timestamp, actor, principal, src_ip, user_agent, operation, resource, target, policy, role, result
logic:
  from cloud_audit within last 24h
  where operation contains any of [AttachPolicy, AttachRolePolicy, AttachUserPolicy, SetIamPolicy, AddRoleAssignment, CreateRoleAssignment]
  and policy or role or target contains any of [Admin, Administrator, Owner, Privileged]
  project timestamp, actor, principal, src_ip, user_agent, operation, resource, target, policy, role, result
correlate:
  join with auth by actor/principal within 24h to identify new device, unfamiliar IP, MFA anomalies, or unusual session.
