# Generic Cloud Storage Public Access Hunt
telemetry: cloud_audit, auth, proxy
fields: timestamp, actor, principal, src_ip, user_agent, operation, resource, target, policy, role, result
logic:
  from cloud_audit within last 24h
  where operation indicates bucket/container/object ACL or policy update
  and target or policy includes public principals [allUsers, anonymous, Everyone, public]
  project timestamp, actor, principal, src_ip, user_agent, operation, resource, target, policy, result
correlate:
  review object read/download/list activity after the public access change and confirm whether exposure was approved.
